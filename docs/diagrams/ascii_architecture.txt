|                                                                                      |
|   from diagrams_as_code import Diagram, Shape                                        |
|   with Diagram("Architecture", theme="nord") as diag:                                |
|       s1 = diag.add(Shape("app", "App Service", icon="server"))                      |
|       s2 = diag.add(Shape("db", "PostgreSQL", icon="database", kind="cylinder"))     |
|       s2.right_of(s1, margin=80)                                                     |
|       s1 >> diag.edge("TCP :5432") >> s2                                             |
|                                                                                      |
|   # or, with no Python driver at all (cli.py):                                       |
|   uv run diagrams-as-code render architecture.py --layout auto --out output          |

|                  LAYOUT PIPELINE  (layout/pipeline.py: run_layout)                   |
|                                                                                      |
|   1. Apply layout-file overrides -- stable id/label-path keys (layout/sidecar.py)    |
|   2. layout="auto": place unpinned shapes/groups per scope, deepest-first            |
|      (layout/graph.py: build_scope_graph, layout/layered.py: LayeredLayout)          |
|   3. Constraint solve -- DAG, cycle detection, right_of/below/above/left_of          |
|      (constraints.py)                                                                |
|   4. Group bounds bottom-up + viewport auto-framing (negative bounds -> 60, 80)      |
|   5. Canvas grows to fit any content past width/height; LayoutWarning on a           |
|      pinned/auto-placed overlap                                                      |
